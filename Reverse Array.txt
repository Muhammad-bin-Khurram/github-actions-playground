#include <iostream>
using namespace std;

void reverse(int arr[], int size){
    for (int i = 0; i < size / 2; i++) {
        int temp = arr[i];
        arr[i] = arr[size - i - 1];
        arr[size - i - 1] = temp;
    }
}

void print(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

int main() {
    int arr[7] = {7,6,5,4,3,2,1};
    cout << "It is your original array: " << endl;
    print(arr, 7);
    
    cout << endl;

    cout << "It is your reversed array: " << endl;

    reverse(arr, 7);
        
    print(arr, 7);
    return 0;
}
